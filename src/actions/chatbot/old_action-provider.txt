import React, {
  JSXElementConstructor,
  ReactElement,
  useContext,
  useEffect,
  useState
} from 'react';
import { createClientMessage } from 'react-chatbot-kit';
import { ChatBotContext } from './ChatBotContext';

import axios, { AxiosResponse } from 'axios';
import { usePusher } from '../_shared/Hooks/usePusher';

axios.defaults.headers.common['Accept'] = 'application/json';
axios.defaults.headers.common['Content-Type'] = 'application/json';

interface ActionProviderProps {
  createChatBotMessage: any;
  setState: any;
  children: React.ReactNode;
  message: string;
}

const ActionProvider = ({
  createChatBotMessage,
  setState,
  children,
  message,
  ...props
}: ActionProviderProps): any => {
  const childElements = React.Children.toArray(children) as ReactElement[];
  const { botData, setBotData } = useContext(ChatBotContext);
  const store = botData?.store;
  const customer = botData?.customer;
  const sessionId = botData?.request_session;
  console.log('customer in Action Provider', customer);
  console.log('botData in Action Provider', botData);

  const recommendedProducts = botData?.recommended_products;

  // ====================== Pusher Connection ======================

  const pusher = usePusher();
  const store_id = store?.id;
  const session = botData.request_session;
  const [aiUserTestResponse, setAiUserTestResponse] = useState('Loading ...');
  const [loadingState, setLoadingState] = useState(false);

  useEffect(() => {
    const subscription = `chat-stream-${store_id}-${session}`;
    // console.log('user channel ID', subscription);
    let channel: any;

    if (
      //@ts-ignore
      window.Echo
    ) {
      // console.log('user channel is in if:', subscription);
      channel =
        //@ts-ignore
        window.Echo.private(subscription).listenToAll((e, data) => {
          console.log('chatbot response:', e);
          // console.log('text:', data.text);s

          if (data?.completed === false) {
            setAiUserTestResponse((prevResponse) =>
              prevResponse === 'Loading ...'
                ? data.text
                : prevResponse + data.text
            );
          } else if (data?.completed === true) {
            setAiUserTestResponse('Loading ...');
            setLoadingState(false);
          }
        });
    }

    return () => {
      channel?.unsubscribe();
    };
  }, [pusher]);

  // =========================== End Pusher Connection ======================

  // ====================== Use Effect - Appending Stream Text ======================
  useEffect(() => {
    if (loadingState) {
      setState((prev: any) => ({
        messages: prev.messages.map((msg: any, index: number) =>
          index === prev.messages.length - 1
            ? {
                ...msg,
                message: aiUserTestResponse
              }
            : msg
        )
      }));
    }
  }, [aiUserTestResponse]);

  console.log('aiUserTestResponse:', aiUserTestResponse);

  // ===============================================================

  //#region  //*=========== Handle Default Message ===========
  const handleDefault = async (messageToSend: string) => {
    setLoadingState(true);

    // Create a loading message - Default message is "Loading..."
    const loadingMessage = createChatBotMessage(aiUserTestResponse, {
      loading: true,
      delay: 0,
      withAvatar: true
    });

    setState((prev: any) => ({
      ...prev,
      messages: [...prev.messages, loadingMessage]
    }));

    // Define the URL and overrideConfig for the fetch call
    // let fetchURL, overrideConfig;
    // console.log('store.name', store?.name);
    // If the store is the Demo ApothecformattedMessage1 are store, use the Apothecare chatflow id, and store data namespace
    // if (botData?.store?.name === 'Apothecare') {
    // fetchURL = `https://copymuze.com/api/ai/ask-chat`;

    try {
      const response: AxiosResponse = await axios.post('/api/ask', {
        question: messageToSend,
        store_id: store?.id,
        customer_id: customer?.id ?? null,
        req_session: sessionId,
        greeting: false
      });
      // console.log('response in action provider ask', response?.data);

      // if (response?.data?.context) {
      //   if (
      //     response?.data?.context === 'other' ||
      //     response?.data?.context === 'address' ||
      //     response?.data?.context === 'customer_service' ||
      //     response?.data?.context === 'help'
      //   ) {
      //     console.log('response.data.message', response?.data?.message);
      //     handleDirectResponse(response?.data?.message);
      //   }
      //   //add the context to the botData
      //   setBotData((prevState) => ({
      //     ...prevState,
      //     context: response.data.context
      //   }));
      // }
    } catch (error) {
      console.error('Error occurred during API call:', error);
    }
  };

  // #endregion  //*======== Handle Default Message ===========

  return (
    <div>
      {childElements.map(
        (child: ReactElement<any, string | JSXElementConstructor<any>>) => {
          return React.cloneElement(child, {
            actions: {
              handleDefault
            }
          });
        }
      )}
    </div>
  );
};

export default ActionProvider;
